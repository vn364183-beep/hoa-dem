""" 🌷 VƯỜN TULIP ĐÊM - bản WOW (chỉ dùng tkinter có sẵn, KHÔNG cần cài gì thêm) Điều khiển: - Click chuột trái : trồng 1 cây tulip màu ngẫu nhiên - Giữ chuột trái + kéo : trồng cả một hàng hoa theo đường kéo - Click chuột phải : bắn pháo hoa 🎆 - Rê chuột qua hoa : hoa nghiêng né theo "gió" từ chuột - Phím SPACE : trồng nguyên một vườn hoa ngẫu nhiên - Phím C : xoá hết hoa """ import tkinter as tk import random import math import colorsys import time WIDTH, HEIGHT = 1100, 700 MAX_TULIPS = 50 root = tk.Tk() root.title("Vườn Tulip Đêm") root.configure(bg="black") canvas = tk.Canvas(root, width=WIDTH, height=HEIGHT, bg="black", highlightthickness=0) canvas.pack(fill="both", expand=True) # ======================= Hàm hỗ trợ ======================= def clamp(x, a=0.0, b=1.0): return max(a, min(b, x)) def ease_out(x): return 1 - (1 - x) ** 3 def ease_in_out(x): return 2 * x * x if x < 0.5 else 1 - (-2 * x + 2) ** 2 / 2 def ease_out_back(x): c1 = 1.70158 c3 = c1 + 1 return 1 + c3 * (x - 1) ** 3 + c1 * (x - 1) ** 2 def hsl_rgb(h, s, l): r, g, b = colorsys.hls_to_rgb(h % 1.0, l, s) return int(r * 255), int(g * 255), int(b * 255) def rgb_hex(rgb, k=1.0): r, g, b = (max(0, min(255, int(c * k))) for c in rgb) return f"#{r:02x}{g:02x}{b:02x}" def transform(points, angle, scale, ox, oy): """Xoay + phóng to các điểm quanh gốc (0,0) rồi dời tới (ox, oy).""" c, s = math.cos(angle), math.sin(angle) out = [] for x, y in points: x *= scale y *= scale out += [ox + x * c - y * s, oy + x * s + y * c] return out def petal_shape(w, h): """Hình cánh hoa tulip (gốc ở (0,0), hướng lên trên).""" return [(0, 0), (w * 0.7, -h * 0.2), (w, -h * 0.55), (w * 0.55, -h * 0.95), (0, -h * 0.82), (-w * 0.55, -h * 0.95), (-w, -h * 0.55), (-w * 0.7, -h * 0.2)] def size(): return max(canvas.winfo_width(), 100), max(canvas.winfo_height(), 100) mouse = {"x": -9999, "y": -9999} # ======================= Bầu trời: trăng + sao ======================= def make_moon(): mx, my = WIDTH - 140, 110 for r, k in ((90, 0.04), (70, 0.07), (55, 0.12), (45, 0.2)): # quầng sáng canvas.create_oval(mx - r, my - r, mx + r, my + r, fill=rgb_hex((200, 210, 255), k), outline="", tags="sky") canvas.create_oval(mx - 32, my - 32, mx + 32, my + 32, fill="#f4f1de", outline="", tags="sky") for dx, dy, r in ((-10, -8, 7), (12, 6, 5), (-4, 14, 4)): # miệng núi lửa canvas.create_oval(mx + dx - r, my + dy - r, mx + dx + r, my + dy + r, fill="#dcd8c0", outline="", tags="sky") stars = [] def make_stars(): for _ in range(220): x, y = random.uniform(0, 1920), random.uniform(0, 1080) * random.random() # dày ở trên r = random.choice((0.6, 0.8, 1.0, 1.0, 1.5)) item = canvas.create_oval(x - r, y - r, x + r, y + r, fill="#000000", outline="", tags="sky") stars.append([item, random.uniform(0.3, 1.0), random.uniform(1, 4), random.uniform(0, 6.28)]) make_stars() make_moon() def update_stars(now): for item, base, speed, ph in stars: k = base * (0.55 + 0.45 * math.sin(now * speed + ph)) canvas.itemconfig(item, fill=rgb_hex((230, 235, 255), k)) # ======================= Cây Tulip ======================= class Tulip: def __init__(self, x, y, hue=None): self.x, self.y = x, y self.t = 0.0 self.lean = 0.0 self.tag = f"tulip{id(self)}" self.phase = random.uniform(0, math.tau) self.next_drop = random.uniform(5, 14) _, h = size() depth = 0.55 + 0.6 * (y / h) # gần thì to, xa thì nhỏ self.size = random.uniform(0.85, 1.15) * depth self.h = random.uniform(120, 200) * depth self.bend = random.uniform(-25, 25) * depth hue = random.random() if hue is None else hue sat = random.uniform(0.75, 1.0) lig = random.uniform(0.5, 0.6) self.rgb = hsl_rgb(hue, sat, lig) self.c_main = rgb_hex(self.rgb) self.c_dark = rgb_hex(self.rgb, 0.65) self.c_light = rgb_hex(hsl_rgb(hue, sat, min(lig * 1.25, 0.82))) self.c_vein = rgb_hex(hsl_rgb(hue, sat, min(lig * 1.45, 0.9))) g = hsl_rgb(0.33 + random.uniform(-0.04, 0.04), 0.6, 0.35) self.c_stem = rgb_hex(g) self.c_leaf = rgb_hex(g, 1.1) self.c_leaf_dark = rgb_hex(g, 0.75) self.head = (x, y) def step(self, dt, now): self.t += dt # "Gió" từ chuột: hoa gần chuột nghiêng tránh ra hx, hy = self.head dx, dy = hx - mouse["x"], hy - mouse["y"] d = math.hypot(dx, dy) target = 0.0 if d < 160: target = (1 if dx >= 0 else -1) * (1 - d / 160) * 45 * self.size self.lean += (target - self.lean) * min(1.0, dt * 5) # Hoa nở xong thỉnh thoảng rụng một cánh bay theo gió if self.t > 3.5: self.next_drop -= dt if self.next_drop <= 0: self.next_drop = random.uniform(6, 16) falling.append(FallingPetal(hx, hy - 15 * self.size, self.rgb, self.size)) def geometry(self, now): g = ease_out(clamp(self.t / 1.5)) sway = math.sin(now * 1.5 + self.phase) * 4 * g + self.lean * g top_x = self.x + self.bend * g + sway top_y = self.y - self.h * g + abs(self.lean) * 0.25 * g return g, sway, top_x, top_y def draw_glow(self, now): """Quầng sáng phía sau bông hoa (vẽ dưới tất cả các cây).""" bloom = ease_in_out(clamp((self.t - 1.8) / 1.5)) if bloom <= 0: return _, _, tx, ty = self.geometry(now) pulse = 0.75 + 0.25 * math.sin(now * 2 + self.phase) cy = ty - 22 * self.size for r, k in ((70, 0.07), (50, 0.12), (32, 0.2)): r *= self.size * (0.9 + 0.1 * pulse) canvas.create_oval(tx - r, cy - r, tx + r, cy + r, outline="", fill=rgb_hex(self.rgb, k * bloom * pulse), tags="dyn") def draw(self, now): t = self.t x, y = self.x, self.y g, sway, top_x, top_y = self.geometry(now) self.head = (top_x, top_y) mid_x = x + self.bend * 0.3 * g + sway * 0.35 mid_y = y - self.h * g * 0.5 # Lá for i, side in enumerate((-1, 1)): ls = ease_out_back(clamp((t - 0.3 - i * 0.2) / 1.2)) if ls <= 0.01: continue L = self.h * 0.55 * ls w = L * 0.22 leaf = [(0, 0), (side * w, -L * 0.35), (side * L * 0.4, -L), (side * w * 0.1, -L * 0.45)] canvas.create_polygon(transform(leaf, sway * 0.004, 1, x, y), smooth=True, fill=self.c_leaf if side > 0 else self.c_leaf_dark, outline="", tags="dyn") if g > 0.01: canvas.create_line(x, y, mid_x, mid_y, top_x, top_y, smooth=True, width=max(2, 4 * self.size), fill=self.c_stem, capstyle="round", tags="dyn") # Bông hoa hs = ease_out_back(clamp((t - 0.8) / 1.0)) * self.size if hs > 0.01: bloom = ease_in_out(clamp((t - 1.8) / 1.5)) tilt = self.bend * 0.004 + sway * 0.012 pw, ph = 18, 46 canvas.create_polygon(transform(petal_shape(pw, ph), tilt, hs, top_x, top_y + 4 * hs), smooth=True, fill=self.c_dark, outline="", tags="dyn") open_angle = 0.1 + bloom * 0.5 for sgn in (-1, 1): canvas.create_polygon( transform(petal_shape(pw, ph), tilt + sgn * open_angle, hs, top_x + sgn * 4 * hs, top_y + 4 * hs), smooth=True, fill=self.c_main, outline="", tags="dyn") canvas.create_polygon( transform(petal_shape(pw * 0.85, ph * (0.95 - bloom * 0.08)), tilt, hs, top_x, top_y + 5 * hs), smooth=True, fill=self.c_light, outline="", tags="dyn") # gân sáng giữa cánh trước vein = transform([(0, -4), (0, -ph * 0.7)], tilt, hs, top_x, top_y + 5 * hs) canvas.create_line(vein, fill=self.c_vein, width=max(1, 1.5 * hs), tags="dyn") # ======================= Cánh hoa rơi ======================= class FallingPetal: def __init__(self, x, y, rgb, sz): self.x, self.y, self.rgb, self.sz = x, y, rgb, sz self.vx = random.uniform(-15, 15) self.vy = random.uniform(10, 25) self.ang = random.uniform(0, 6.28) self.spin = random.uniform(-3, 3) self.life = 1.0 self.phase = random.uniform(0, 6.28) def step(self, dt, now): self.x += (self.vx + math.sin(now * 1.7 + self.phase) * 30) * dt self.y += self.vy * dt self.ang += self.spin * dt self.life -= dt / 7 return self.life > 0 def draw(self): pts = transform(petal_shape(7, 16), self.ang, self.sz, self.x, self.y) canvas.create_polygon(pts, smooth=True, fill=rgb_hex(self.rgb, min(1, self.life * 1.5)), outline="", tags="dyn") # ======================= Đom đóm ======================= class Firefly: def __init__(self): w, h = size() self.x, self.y = random.uniform(0, w), random.uniform(h * 0.3, h) self.vx, self.vy = random.uniform(-20, 20), random.uniform(-20, 20) self.phase = random.uniform(0, 6.28) self.speed = random.uniform(1.5, 3) def step(self, dt, now): w, h = size() self.vx = clamp(self.vx + random.uniform(-60, 60) * dt, -35, 35) self.vy = clamp(self.vy + random.uniform(-60, 60) * dt, -35, 35) self.x = (self.x + self.vx * dt) % w self.y += self.vy * dt if self.y < h * 0.25 or self.y > h - 5: self.vy *= -1 def draw(self, now): b = max(0.0, math.sin(now * self.speed + self.phase)) ** 2 if b < 0.03: return col = (190, 255, 110) for r, k in ((9, 0.12), (5, 0.35), (2, 1.0)): canvas.create_oval(self.x - r, self.y - r, self.x + r, self.y + r, fill=rgb_hex(col, k * b), outline="", tags="dyn") # ======================= Hạt / pháo hoa / sao băng ======================= particles = [] # [x, y, vx, vy, life, rgb, decay, gravity, size] rings = [] # [x, y, life, rgb] rockets = [] # [x, y, tx, ty, rgb] meteors = [] # [x, y, vx, vy, life] def burst(x, y, rgb): rings.append([x, y, 1.0, rgb]) for _ in range(30): a = random.uniform(math.pi * 1.1, math.pi * 1.9) sp = random.uniform(60, 220) particles.append([x, y, math.cos(a) * sp, math.sin(a) * sp, 1.0, rgb, 1 / 1.4, 20, 2.5]) def explode(x, y): hue = random.random() multi = random.random() < 0.4 n = 90 for i in range(n): a = i / n * math.tau + random.uniform(-0.05, 0.05) sp = random.uniform(120, 260) col = hsl_rgb(random.random() if multi else hue + random.uniform(-0.05, 0.05), 1, 0.6) particles.append([x, y, math.cos(a) * sp, math.sin(a) * sp, 1.0, col, 1 / 1.8, 90, 2.8]) for _ in range(25): # tia lấp lánh trắng a = random.uniform(0, math.tau) sp = random.uniform(30, 120) particles.append([x, y, math.cos(a) * sp, math.sin(a) * sp, 1.0, (255, 255, 230), 1 / 1.2, 40, 1.8]) def launch_firework(tx, ty): _, h = size() rockets.append([tx + random.uniform(-40, 40), h, tx, ty, (255, 220, 160)]) def step_effects(dt): # tên lửa pháo hoa for r in rockets[:]: r[0] += (r[2] - r[0]) * min(1, dt * 3) r[1] -= 650 * dt particles.append([r[0], r[1], random.uniform(-15, 15), random.uniform(20, 60), 0.6, r[4], 1 / 0.5, 0, 1.6]) if r[1] <= r[3]: rockets.remove(r) explode(r[2], r[3]) # hạt for p in particles[:]: p[4] -= dt * p[6] if p[4] <= 0: particles.remove(p) continue p[2] *= 0.97 p[3] = p[3] * 0.97 + p[7] * dt p[0] += p[2] * dt p[1] += p[3] * dt for r in rings[:]: r[2] -= dt if r[2] <= 0: rings.remove(r) # sao băng if random.random() < dt / 4: w, h = size() meteors.append([random.uniform(w * 0.1, w), random.uniform(0, h * 0.25), -random.uniform(500, 800), random.uniform(200, 350), 1.0]) for m in meteors[:]: m[0] += m[2] * dt m[1] += m[3] * dt m[4] -= dt / 1.2 if m[4] <= 0: meteors.remove(m) def draw_effects(): for m in meteors: for seg, k in ((0.12, 1.0), (0.25, 0.45), (0.4, 0.15)): canvas.create_line(m[0], m[1], m[0] - m[2] * seg, m[1] - m[3] * seg, fill=rgb_hex((220, 230, 255), k * m[4]), width=2 * k + 0.5, capstyle="round", tags="dyn") for r in rings: rad = (1 - r[2]) * 80 canvas.create_oval(r[0] - rad, r[1] - rad * 0.3, r[0] + rad, r[1] + rad * 0.3, outline=rgb_hex(r[3], r[2]), width=2, tags="dyn") for x, y, vx, vy, life, rgb, _, _, s in particles: s = s * life + 0.5 canvas.create_oval(x - s, y - s, x + s, y + s, fill=rgb_hex(rgb, life * 1.3), outline="", tags="dyn") for r in rockets: canvas.create_oval(r[0] - 2.5, r[1] - 2.5, r[0] + 2.5, r[1] + 2.5, fill="#fff3d0", outline="", tags="dyn") # ======================= Tương tác ======================= tulips = [] falling = [] fireflies = [Firefly() for _ in range(28)] hint = canvas.create_text(WIDTH / 2, HEIGHT - 30, fill="#666666", font=("Segoe UI", 12), text="🌷 Click: trồng hoa · Kéo chuột: trồng cả hàng · Chuột phải: pháo hoa" " · SPACE: cả vườn · C: xoá") counter = canvas.create_text(20, 20, anchor="nw", fill="#777777", font=("Segoe UI", 12), text="") last_plant = {"x": -999, "y": -999} def plant(x, y, hue=None): t = Tulip(x, y, hue) tulips.append(t) tulips.sort(key=lambda tl: tl.y) # cây xa vẽ trước, cây gần vẽ sau if len(tulips) > MAX_TULIPS: tulips.remove(max(tulips, key=lambda tl: tl.t)) # bỏ cây già nhất burst(x, y, t.rgb) canvas.itemconfig(hint, state="hidden") def on_click(e): plant(e.x, e.y) last_plant.update(x=e.x, y=e.y) def on_drag(e): mouse.update(x=e.x, y=e.y) if math.hypot(e.x - last_plant["x"], e.y - last_plant["y"]) > 40: on_click(e) def on_move(e): mouse.update(x=e.x, y=e.y) def on_leave(e): mouse.update(x=-9999, y=-9999) def garden(e=None): w, h = size() base_hue = random.random() for i in range(14): x = random.uniform(w * 0.05, w * 0.95) y = random.uniform(h * 0.5, h * 0.97) hue = base_hue + random.uniform(-0.15, 0.15) if random.random() < 0.6 else None root.after(i * 120, lambda x=x, y=y, hue=hue: plant(x, y, hue)) def clear(e=None): tulips.clear() falling.clear() canvas.bind("", on_click) canvas.bind("", on_drag) canvas.bind("", on_move) canvas.bind("", on_leave) canvas.bind("", lambda e: launch_firework(e.x, e.y)) root.bind("", garden) root.bind("", clear) root.bind("", clear) # ======================= Vòng lặp ======================= last = time.perf_counter() def loop(): global last now = time.perf_counter() dt = min(now - last, 0.05) last = now update_stars(now) for t in tulips: t.step(dt, now) falling[:] = [p for p in falling if p.step(dt, now)] for f in fireflies: f.step(dt, now) step_effects(dt) canvas.delete("dyn") for t in tulips: # quầng sáng nằm dưới tất cả các cây t.draw_glow(now) for t in tulips: t.draw(now) for p in falling: p.draw() for f in fireflies: f.draw(now) draw_effects() canvas.itemconfig(counter, text=f"🌷 {len(tulips)}" if tulips else "") canvas.tag_raise(counter) canvas.tag_raise(hint) root.after(16, loop) loop() root.mainloop()
